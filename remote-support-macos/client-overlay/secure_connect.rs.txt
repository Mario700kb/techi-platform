// techi-secure-connect-stdin-v1
#[cfg(all(target_os = "macos", feature = "flutter"))]
#[used]
#[no_mangle]
pub static TECHI_SECURE_CONNECT_CONTRACT: [u8; 29] = *b"techi-secure-connect-stdin-v1";

#[cfg(all(target_os = "macos", feature = "flutter"))]
fn techi_secure_connect_from_stdin(mut args: std::env::Args) -> Option<Vec<String>> {
    use std::io::{Read, Write};

    std::hint::black_box(&TECHI_SECURE_CONNECT_CONTRACT);
    let _exe = args.next();
    let command = args.next().unwrap_or_default();
    let self_test = command == "--techi-connect-self-test";
    let remote_id = if self_test {
        "000000000".to_owned()
    } else if command == "--connect" {
        args.next().unwrap_or_default()
    } else {
        return None;
    };
    if !self_test && args.next().as_deref() != Some("--techi-connect-stdin") {
        return None;
    }
    if self_test && args.next().is_some() {
        return None;
    }
    if remote_id.len() < 6
        || remote_id.len() > 64
        || !remote_id
            .bytes()
            .all(|value| value.is_ascii_alphanumeric() || value == b'_' || value == b'-')
    {
        return None;
    }

    let mut credential = Vec::new();
    if std::io::stdin()
        .take(513)
        .read_to_end(&mut credential)
        .is_err()
        || credential.is_empty()
        || credential.len() > 512
    {
        credential.fill(0);
        return None;
    }
    if self_test {
        credential.fill(0);
        println!("TECHI_CONNECT_ACCEPTED_V1");
        let _ = std::io::stdout().flush();
        return None;
    }

    let mut password = match String::from_utf8(credential) {
        Ok(value) if !value.is_empty() => value,
        Ok(mut value) => {
            unsafe { value.as_mut_vec().fill(0) };
            return None;
        }
        Err(error) => {
            let mut bytes = error.into_bytes();
            bytes.fill(0);
            return None;
        }
    };
    let delivered = crate::ipc::send_techi_connect(&remote_id, &password).is_ok();
    println!("TECHI_CONNECT_ACCEPTED_V1");
    let _ = std::io::stdout().flush();
    if delivered {
        unsafe { password.as_mut_vec().fill(0) };
        None
    } else {
        Some(vec![
            "--connect".to_owned(),
            remote_id,
            "--password".to_owned(),
            password,
        ])
    }
}
